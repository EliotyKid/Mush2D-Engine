cmake_minimum_required(VERSION 3.24)
project(Mush2DEngine LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Vulkan REQUIRED COMPONENTS glslc)
include(cmake/Dependencies.cmake)
include(cmake/Shaders.cmake)

file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS src/engine/*.cpp)
file(GLOB_RECURSE GAME_SOURCES CONFIGURE_DEPENDS src/game/*.cpp)

add_library(mush2d_engine STATIC ${ENGINE_SOURCES})
# external/ only holds stb_image.h and stb_image_write.h
target_include_directories(mush2d_engine PUBLIC src external)
target_include_directories(mush2d_engine PRIVATE ${miniaudio_SOURCE_DIR})
target_compile_definitions(mush2d_engine PRIVATE MA_NO_FLAC MA_NO_MP3) # audio is .wav only
find_package(Threads REQUIRED)
target_link_libraries(mush2d_engine PRIVATE Threads::Threads ${CMAKE_DL_LIBS}) # miniaudio: threads, and dlopen of the audio backends on Linux
target_compile_definitions(mush2d_engine PUBLIC GLFW_INCLUDE_NONE $<$<CONFIG:Debug>:MUSH2D_SOURCE_DIR="${CMAKE_SOURCE_DIR}">)
target_link_libraries(mush2d_engine PUBLIC glfw glm::glm Vulkan::Vulkan nlohmann_json::nlohmann_json imgui vk-bootstrap::vk-bootstrap GPUOpen::VulkanMemoryAllocator)

# Every game object header goes into a generated all_objects.cpp, so its MUSH2D_OBJECT registration exists
# in the game library. mush2dRegisterGameTypes() (called by main and the tests) keeps that file linked.
file(GLOB GAME_OBJECT_HEADERS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/game/objects/*.hpp)
set(MUSH2D_OBJECT_INCLUDES "")
foreach(header ${GAME_OBJECT_HEADERS})
    file(RELATIVE_PATH header_rel ${CMAKE_SOURCE_DIR}/src ${header})
    string(APPEND MUSH2D_OBJECT_INCLUDES "#include \"${header_rel}\"\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/generated/all_objects.cpp CONTENT
"// Generated by CMakeLists.txt from src/game/objects/*.hpp. Do not edit.
#include \"game/RegisterGameTypes.hpp\"
@MUSH2D_OBJECT_INCLUDES@
void mush2dRegisterGameTypes() {} // the includes above register the types; calling this keeps them linked
" @ONLY)

add_library(mush2d_game STATIC ${GAME_SOURCES} ${CMAKE_BINARY_DIR}/generated/all_objects.cpp)
target_link_libraries(mush2d_game PUBLIC mush2d_engine)

add_executable(Mush2DEngine src/main.cpp)
target_link_libraries(Mush2DEngine PRIVATE mush2d_game)
mush2d_compile_shaders(Mush2DEngine)
mush2d_copy_assets(Mush2DEngine)

enable_testing()
add_executable(mush2d_tests tests/tests.cpp)
target_link_libraries(mush2d_tests PRIVATE mush2d_game)
target_compile_definitions(mush2d_tests PRIVATE MUSH2D_TEST_DATA="${CMAKE_SOURCE_DIR}/tests/data")
if(MSVC)
    target_compile_options(mush2d_tests PRIVATE /bigobj) # one large test file
endif()
add_test(NAME mush2d_tests COMMAND mush2d_tests)
